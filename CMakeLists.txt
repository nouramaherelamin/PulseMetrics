cmake_minimum_required(VERSION 3.20)
project(PulseMetrics VERSION 2.0 LANGUAGES C)
if(NOT WIN32)
  message(FATAL_ERROR "PulseMetrics requires Windows and the UCRT64 toolchain")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS database/schema.sql)
file(READ "${CMAKE_SOURCE_DIR}/database/schema.sql" SCHEMA_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1," SCHEMA_BYTES "${SCHEMA_HEX}")
configure_file(native/collector/schema.h.in "${CMAKE_BINARY_DIR}/generated/pm_schema.h" @ONLY)
add_library(pulse_sqlite STATIC native/vendor/sqlite3.c)
target_include_directories(pulse_sqlite PUBLIC native/vendor)
target_compile_definitions(pulse_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION=1)
add_library(pulse_json STATIC native/vendor/cJSON.c)
target_include_directories(pulse_json PUBLIC native/vendor)
add_library(pulsecore STATIC native/core/src/clock_v2.c native/core/src/protocol_v2.c
  native/core/src/store_v2.c native/core/src/json_v2.c native/core/src/persistence_v2.c)
target_include_directories(pulsecore PUBLIC native/core/include)
target_include_directories(pulsecore PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(pulsecore PUBLIC pulse_sqlite pulse_json ws2_32)
target_compile_definitions(pulsecore PRIVATE _CRT_SECURE_NO_WARNINGS)
target_compile_options(pulsecore PRIVATE -Wall -Wextra -Wpedantic)
add_executable(pulsecollector_v2 native/collector/src/engine_v2.c)
target_link_libraries(pulsecollector_v2 PRIVATE pulsecore)
target_compile_options(pulsecollector_v2 PRIVATE -Wall -Wextra)
add_executable(pulsemonitor_v2 native/monitor/src/monitor_v2.c)
target_link_libraries(pulsemonitor_v2 PRIVATE pulsecore psapi)
target_compile_options(pulsemonitor_v2 PRIVATE -Wall -Wextra)
add_executable(pulse_sender native/sender/main.c)
target_link_libraries(pulse_sender PRIVATE pulsecore)
target_compile_options(pulse_sender PRIVATE -Wall -Wextra)
if(BUILD_TESTING)
  add_executable(test_persistence_v2 native/tests/test_persistence_v2.c)
  target_link_libraries(test_persistence_v2 PRIVATE pulsecore)
  target_compile_options(test_persistence_v2 PRIVATE -UNDEBUG -Wall -Wextra)
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/test-run/persistence")
  add_test(NAME v2_persistence COMMAND test_persistence_v2)
  set_tests_properties(v2_persistence PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/test-run/persistence" TIMEOUT 30)
  add_executable(test_contract native/tests/test_contract.c)
  target_link_libraries(test_contract PRIVATE pulsecore)
  target_compile_options(test_contract PRIVATE -UNDEBUG -Wall -Wextra)
  add_test(NAME v2_contract COMMAND test_contract)
endif()
